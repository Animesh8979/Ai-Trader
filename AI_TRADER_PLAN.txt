  ```python
  import numpy as np
  from scipy.stats import skew, kurtosis, norm

  def cornish_fisher_var(returns, confidence_level=0.99):
      z = norm.ppf(1 - confidence_level)
      s = skew(returns)
      k = kurtosis(returns)  # Excess kurtosis
      cf_z = z + (z**2 - 1)*s/6 + (z**3 - 3*z)*k/24 - (2*z**3 - 5*z)*(s**2)/36
      return -(np.mean(returns) + cf_z * np.std(returns))
  ```

  File: D:\Ai trader\src\godmode\execution\live_runner.py
  ```python
  import numpy as np

  class MirroredDoubleBuffer:
      def __init__(self, capacity=1000):
          self.capacity = capacity
          self.buffer = np.zeros(2 * capacity)
          self.index = 0

      def append(self, val):
          idx = self.index % self.capacity
          self.buffer[idx] = val
          self.buffer[idx + self.capacity] = val
          self.index += 1

      def get_window(self, size):
          end = (self.index % self.capacity) + self.capacity
          return self.buffer[end - size : end]
  ```
